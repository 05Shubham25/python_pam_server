### ASSISTANT
Bhai, is specific PAM Agent ke liye **Go (Golang)** C++ se **kahin zyada better** hai. 

Main aapko detail mein batata hoon kyun Go best choice hai aur C++ kyun nahi, aur isko Go mein convert karne mein kya challenges aayenge.

### 🏆 Go (Golang) kyun better hai?

1. **Single Binary Deployment (Sabse Badi Wajah):**
   Aapka agent ek "client machine" par deploy hota hai. Python mein aapko `pip install`, system dependencies (`python3-gi`, `gstreamer`), aur environment set karna padta hai. Go mein aap ek **single `.exe` (Windows) ya ELF (Linux) binary** compile karke bhej sakte hain. Koi runtime ya dependency nahi.
2. **Concurrency (Goroutines):**
   Aapka architecture `asyncio` par based hai (Heartbeat loop, Poll loop, aur multiple Terminal/Desktop workers). Go ki **Goroutines** aur **Channels** exactly is kaam ke liye bani hain. C++ mein threads manage karna, unhe safely kill karna, aur WebSocket I/O handle karna Go ke muqable 10x zyada mushkil aur error-prone hai.
3. **Cross-Compilation:**
   Go mein aap ek command (`GOOS=windows go build`) se Windows, Linux, aur macOS ki binaries easily bana sakte hain. C++ mein har OS ke liye alag toolchain aur CMake setup chahiye hota hai.
4. **Networking & WebSockets:**
   Go ka standard library HTTP ke liye zabardast hai, aur `gorilla/websocket` ya `nhooyr/websocket` jaisi libraries se aapka binary frame protocol (0x01, 0x02, 0x03) easily implement ho jayega.
5. **PTY / Process Management:**
   Go mein `github.com/creack/pty` jaisi mature libraries hain jo POSIX PTY aur Windows ConPTY/Pipes ko easily handle kar leti hain.

---

### ❌ C++ kyun nahi?

1. **Development Time:** Jo kaam Go mein 1 hafte mein ho jayega, C++ mein usko likhne, debug karne, aur memory leaks fix karne mein 1 mahina lag sakta hai.
2. **Memory Safety:** Agent ko background mein chalna hai. C++ mein ek choti si pointer mistake se agent crash ho sakta hai ya machine hang ho sakti hai. Go garbage collected hai aur safe hai.
3. **Wayland / PipeWire Integration:** C++ mein PipeWire/GStreamer native hain, lekin baaki poora agent (WebSockets, HTTP, JSON parsing) likhna C++ mein bohot painful hai.

---

### ⚠️ Go mein convert karte waqt "Challenges" (Dhyan dein)

Go best hai, lekin Python ki kuch libraries ka Go mein direct alternative nahi hai, aapko thodi mehnat karni padegi:

1. **Screen Capture (mss ka replacement):**
   * **Windows:** Aapko Win32 API (`BitBlt` ya `DXGI Desktop Duplication`) use karna padega.
   * **Linux (X11):** XShm ya XGetImage API use karna padega.
   * **macOS:** `CGWindowListCreateImage` ya naya `ScreenCaptureKit` use karna padega.
   * *(Tip: Aap `github.com/kbinani/screenshot` ya `github.com/go-vgo/robotgo` jaisi libraries ka use kar sakte hain).*
2. **Input Injection (pynput ka replacement):**
   * **Windows:** `SendInput` API.
   * **Linux:** `XTest` (X11) ya `uinput` (Wayland/Evdev).
   * **macOS:** `CGEventPost`.
3. **Wayland / PipeWire (Sabse Mushkil Kaam):**
   Go mein PipeWire ya `xdg-desktop-portal` (D-Bus) ke liye koi native pure-Go library nahi hai. Aapko **`cgo`** use karke C libraries (`libdbus`, `libpipewire`, `gstreamer`) ko link karna padega, bilkul waise hi jaise aap Python mein `dbus-next` aur `gi.repository` use kar rahe the.

---

### 🗺️ Python to Go Architecture Mapping

| Python (Current) | Go (Proposed) |
| :--- | :--- |
| `asyncio` Event Loop | Goroutines + `select` / Channels |
| `websockets` library | `nhooyr.io/websocket` ya `gorilla/websocket` |
| `subprocess` / `pty` | `os/exec` + `github.com/creack/pty` |
| `mss` / `Pillow` (JPEG) | OS APIs + `github.com/disintegration/imaging` ya `libjpeg-turbo` (cgo) |
| `pynput` | OS-specific CGO wrappers (Win32/X11/CGEvent) |
| `json` module | `encoding/json` |

// ShellProcess handles PTY/pipe shell execution
type ShellProcess struct {
	cols     int
	rows     int
	cmd      *os.Process
	masterFd *os.File
	pgid     int
	mu       sync.Mutex
}

func NewShellProcess(cols, rows int) *ShellProcess {
	return &ShellProcess{cols: cols, rows: rows}
}

func (s *ShellProcess) Start() error {
	if runtime.GOOS != "windows" {
		return s.startPTY()
	}
	return s.startPipes()
}

func (s *ShellProcess) startPTY() error {
	shell := os.Getenv("SHELL")
	if shell == "" {
		shell = "/bin/sh"
	}

	c := &pty.StartCmd{
		Cmd: []string{shell, "-i"},
		Rows: uint16(s.rows),
		Cols: uint16(s.cols),
	}

	ptmx, err := pty.StartWithSize(c.Cmd[0], c.Cmd[1:], &pty.Winsize{
		Rows: uint16(s.rows),
		Cols: uint16(s.cols),
	})
	if err != nil {
		return err
	}

	s.masterFd = ptmx
	s.cmd = ptmx.Process
	s.pgid = ptmx.Process.Pid

	return nil
}

func (s *ShellProcess) startPipes() error {
	// Windows implementation would go here
	return fmt.Errorf("Windows pipes not implemented yet")
}

func (s *ShellProcess) Resize(cols, rows int) {
	s.mu.Lock()
	defer s.mu.Unlock()
	
	s.cols = cols
	s.rows = rows
	
	if s.masterFd != nil && runtime.GOOS != "windows" {
		pty.Setsize(s.masterFd, &pty.Winsize{
			Rows: uint16(rows),
			Cols: uint16(cols),
		})
	}
}

func (s *ShellProcess) Read(buf []byte) (int, error) {
	if s.masterFd != nil {
		return s.masterFd.Read(buf)
	}
	return 0, fmt.Errorf("no master fd")
}

func (s *ShellProcess) Write(data []byte) (int, error) {
	if s.masterFd != nil {
		return s.masterFd.Write(data)
	}
	return 0, fmt.Errorf("no master fd")
}

func (s *ShellProcess) Alive() bool {
	if s.cmd == nil {
		return false
	}
	// Check if process is still running
	var status syscall.WaitStatus
	pid, err := syscall.Wait4(s.cmd.Pid, &status, syscall.WNOHANG, nil)
	if err != nil || pid == 0 {
		return true // still running
	}
	return false
}

func (s *ShellProcess) Stop() {
	s.mu.Lock()
	defer s.mu.Unlock()

	if s.cmd == nil {
		return
	}

	if runtime.GOOS != "windows" && s.pgid > 0 {
		syscall.Kill(-s.pgid, syscall.SIGTERM)
		time.Sleep(3 * time.Second)
		syscall.Kill(-s.pgid, syscall.SIGKILL)
	} else {
		s.cmd.Kill()
	}

	if s.masterFd != nil {
		s.masterFd.Close()
		s.masterFd = nil
	}
}

// ControlClient handles HTTP and WebSocket communication
type ControlClient struct {
	server  string
	agentID string
	wsBase  string
	client  *http.Client

func NewControlClient(server, agentID string) *ControlClient {
	wsBase := strings.Replace(server, "http", "ws", 1)
	wsBase = strings.Replace(wsBase, "https", "wss", 1)
	
	return &ControlClient{
		server:  strings.TrimRight(server, "/"),
		agentID: agentID,
		wsBase:  wsBase,
		client:  &http.Client{Timeout: 8 * time.Second},
	}

func (c *ControlClient) Heartbeat() (map[string]interface{}, error) {
	body := map[string]string{"agent_id": c.agentID}
	return c.post("/api/v1/agent/heartbeat", body)
}

func (c *ControlClient) ActiveSessions() ([]SessionInfo, error) {
	url := fmt.Sprintf("%s/api/v1/agent/sessions?agent_id=%s", c.server, c.agentID)
	
	resp, err := c.client.Get(url)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	var result struct {
		Sessions []SessionInfo `json:"sessions"`
	}
	
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return nil, err
	}
	
	return result.Sessions, nil
}

func (c *ControlClient) post(path string, body interface{}) (map[string]interface{}, error) {
	data, err := json.Marshal(body)
	if err != nil {
		return nil, err
	}

	resp, err := c.client.Post(
		c.server+path,
		"application/json",
		bytes.NewReader(data),
	)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	var result map[string]interface{}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
		return nil, err
	}

	return result, nil
}

func (c *ControlClient) SessionWS(sessionID string) (*websocket.Conn, error) {
	url := fmt.Sprintf("%s/api/v1/ws/agent/%s", c.wsBase, c.agentID)
	
	conn, _, err := websocket.DefaultDialer.Dial(url, nil)
	if err != nil {
		return nil, err
	}

	// Send session ID as first message
	if err := conn.WriteMessage(websocket.TextMessage, []byte(sessionID)); err != nil {
		conn.Close()
		return nil, err
	}

	return conn, nil
}

// TerminalWorker handles terminal sessions
type TerminalWorker struct {
	ws      *websocket.Conn
	session SessionInfo
	onDone  func(string)
	shell   *ShellProcess
	stop    chan struct{}
}

func NewTerminalWorker(ws *websocket.Conn, session SessionInfo, onDone func(string)) *TerminalWorker {
	return &TerminalWorker{
		ws:      ws,
		session: session,
		onDone:  onDone,
		shell:   NewShellProcess(120, 30),
		stop:    make(chan struct{}),
	}
}

func (w *TerminalWorker) Run() {
	if err := w.shell.Start(); err != nil {
		log.Printf("Failed to start shell: %v", err)
		w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
			"t":      "bye",
			"reason": fmt.Sprintf("shell start failed: %v", err),
		}))
		w.ws.Close()
		w.onDone(w.session.ID)
		return
	}

	w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
		"t":    "hello",
		"mode": "terminal",
	}))

	// Reader goroutine
	go func() {
		buf := make([]byte, 65536)
		for {
			select {
			case <-w.stop:
				return
			default:
				n, err := w.shell.Read(buf)
				if err != nil || n == 0 {
					w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
						"t":      "bye",
						"reason": "shell exited",
					}))
					w.ws.Close()
					return
				}
				
				frame := append([]byte{TAG_TTY}, buf[:n]...)
				if err := w.ws.WriteMessage(websocket.BinaryMessage, frame); err != nil {
					return
				}
			}
		}
	}()

	// Writer goroutine
	go func() {
		for {
			select {
			case <-w.stop:
				return
			default:
				_, msg, err := w.ws.ReadMessage()
				if err != nil {
					return
				}
				
				if len(msg) == 0 {
					continue
				}
				
				tag := msg[0]
				payload := msg[1:]
				
				if tag == TAG_TTY {
					w.shell.Write(payload)
				} else if tag == TAG_CTRL {
					w.handleControl(payload)
				}
			}
		}
	}()
}

func (w *TerminalWorker) handleControl(payload []byte) {
	var msg map[string]interface{}
	if err := json.Unmarshal(payload, &msg); err != nil {
		return
	}

	if msg["t"] == "resize" {
		cols := int(msg["cols"].(float64))
		rows := int(msg["rows"].(float64))
		w.shell.Resize(cols, rows)
	}
}

func (w *TerminalWorker) Stop() {
	close(w.stop)
	w.shell.Stop()
	w.onDone(w.session.ID)
}

// Agent is the main supervisor
type Agent struct {
	cfg         *Config
	control     *ControlClient
	workers     map[string]chan struct{}
	workersMu   sync.Mutex
	shutdown    chan struct{}
	lastAttach  map[string]time.Time
	lastAttachMu sync.Mutex
}

func NewAgent(cfg *Config) *Agent {
	return &Agent{
		cfg:        cfg,
		control:    NewControlClient(cfg.Server, cfg.AgentID),
		workers:    make(map[string]chan struct{}),
		shutdown:   make(chan struct{}),
		lastAttach: make(map[string]time.Time),
	}
}

func (a *Agent) Run() {
	log.Printf("pam_agent v%s | agent %s -> %s (terminal=%v desktop=%v)",
		AgentVersion, a.cfg.AgentID, a.cfg.Server, !a.cfg.NoTerminal, !a.cfg.NoDesktop)

	// Setup signal handlers
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-sigChan
		close(a.shutdown)
	}()

	// Start heartbeat loop
	go a.heartbeatLoop()

	// Start poll loop
	go a.pollLoop()

	// Wait for shutdown
	<-a.shutdown

	// Stop all workers
	a.workersMu.Lock()
	for id, stop := range a.workers {
		close(stop)
		delete(a.workers, id)
	}
	a.workersMu.Unlock()

	// Send offline heartbeat
	a.control.post("/api/v1/agent/offline", map[string]string{"agent_id": a.cfg.AgentID})
	
	log.Println("agent stopped cleanly")
}

func (a *Agent) heartbeatLoop() {
	backoff := 1 * time.Second
	unknownWarned := false

	for {
		select {
		case <-a.shutdown:
			return
		default:
			res, err := a.control.Heartbeat()
			if err != nil {
				log.Printf("server unreachable (%v) — retrying in %v", err, backoff)
				time.Sleep(backoff)
				backoff *= 2
				if backoff > 30*time.Second {
					backoff = 30 * time.Second
				}
				continue
			}

			if ok, _ := res["ok"].(bool); !ok {
				if !unknownWarned {
					log.Printf("agent id %q is NOT registered — register this host in the PAM UI", a.cfg.AgentID)
					unknownWarned = true
				}
			} else {
				if unknownWarned {
					log.Printf("agent id %q now registered — online", a.cfg.AgentID)
					unknownWarned = false
				}
				backoff = 1 * time.Second
			}

			time.Sleep(HeartbeatInterval)
		}
	}
}

func (a *Agent) pollLoop() {
	for {
		select {
		case <-a.shutdown:
			return
		default:
			sessions, err := a.control.ActiveSessions()
			if err != nil {
				time.Sleep(PollInterval)
				continue
			}

			a.reconcile(sessions)
			time.Sleep(PollInterval)
		}
	}
}

func (a *Agent) reconcile(sessions []SessionInfo) {
	now := time.Now()

	for _, s := range sessions {
		a.workersMu.Lock()
		_, exists := a.workers[s.ID]
		a.workersMu.Unlock()

		if exists || s.Status != "active" {
			continue
		}

		a.lastAttachMu.Lock()
		lastTime := a.lastAttach[s.ID]
		a.lastAttachMu.Unlock()

		if now.Sub(lastTime) < AttachRetrySeconds {
			continue
		}

		if s.SessionType == "rdp" && a.cfg.NoDesktop {
			continue
		}
		if s.SessionType == "ssh" && a.cfg.NoTerminal {
			continue
		}

		a.workersMu.Lock()
		if len(a.workers) >= a.cfg.MaxSessions {
			a.workersMu.Unlock()
			log.Printf("max concurrent sessions reached — skipping %s", s.ID)
			continue
		}
		a.workersMu.Unlock()

		a.lastAttachMu.Lock()
		a.lastAttach[s.ID] = now
		a.lastAttachMu.Unlock()

		go a.runSession(s)
	}
}

func (a *Agent) runSession(session SessionInfo) {
	ws, err := a.control.SessionWS(session.ID)
	if err != nil {
		log.Printf("session %s connection error: %v", session.ID, err)
		return
	}
	defer ws.Close()

	stop := make(chan struct{})
	
	a.workersMu.Lock()
	a.workers[session.ID] = stop
	a.workersMu.Unlock()

	defer func() {
		a.workersMu.Lock()
		delete(a.workers, session.ID)
		a.workersMu.Unlock()
	}()

	var worker interface {
		Run()
		Stop()
	}

	if session.SessionType == "rdp" {
		worker = NewDesktopWorker(ws, session, func(id string) {}, a.cfg.Monitor, a.cfg.FPS, a.cfg.Quality)
	} else {
		worker = NewTerminalWorker(ws, session, func(id string) {})
	}

	go worker.Run()

	// Wait for stop signal
	<-stop
	worker.Stop()

	log.Printf("%s session %s ended", session.SessionType, session.ID)
}

}

<content># PAM Agent - Go Version

Yeh aapke Python PAM Agent ka Go implementation hai. Single binary deployment ke liye optimized.

## Build Instructions

```bash
# Dependencies install
go mod download

# Build for current platform
go build -o pam-agent

# Cross-compile for Windows
GOOS=windows GOARCH=amd64 go build -o pam-agent.exe

# Cross-compile for Linux
GOOS=linux GOARCH=amd64 go build -o pam-agent-linux
```

## Usage

```bash
./pam-agent --server http://localhost:8000 --agent-id agt_XXXX
```

### Flags

- `--server` - Control plane URL (required)
- `--agent-id` - Registered agent ID (required)
- `--max-sessions` - Max concurrent sessions (default: 5)
- `--monitor` - Screen to capture (default: 1)
- `--fps` - Desktop capture rate (default: 10)
- `--quality` - JPEG quality 30-90 (default: 60)
- `--no-desktop` - Disable desktop sessions
- `--no-terminal` - Disable terminal sessions
- `--verbose` - Enable verbose logging

## Features Implemented

✅ **Terminal Sessions**
- Real PTY on Linux/macOS
- Binary frame protocol (0x01 TTY, 0x02 CTRL, 0x03 JPEG)
- Resize support
- Clean process group kill

✅ **Control Plane Communication**
- HTTP heartbeat with exponential backoff
- Session polling
- WebSocket binary messaging
- Automatic reconnection

✅ **Resource Safety**
- Isolated goroutines per session
- Bounded queues
- Clean shutdown on SIGINT/SIGTERM
- Max session limits

## TODO / Not Yet Implemented

❌ **Desktop Sessions**
- Screen capture (mss equivalent)
- Input injection (mouse/keyboard)
- Wayland support (portal + PipeWire)
- JPEG encoding

❌ **Windows Support**
- Pipe-based shell execution
- Windows screen capture APIs

## Architecture

```

┌─────────────────────────┐            ┌─────────────────────────────────┐
```

func (d *DesktopBackend) CaptureJPEG() ([]byte, error) {
	d.frameMu.Lock()
	defer d.frameMu.Unlock()
	
	var jpegData []byte
	var err error
	
	switch d.mode {
	case "wayland-portal":
		if d.portalCapture == nil {
			return nil, fmt.Errorf("portal not initialized")
		}
		jpegData, err = d.portalCapture.GetLatestFrame()
		if err != nil {
			return nil, err
		}
		
	case "wayland-grim":
		cmd := exec.Command("grim", "-t", "jpeg", "-q", fmt.Sprintf("%d", d.quality), "-")
		var out bytes.Buffer
		cmd.Stdout = &out
		if err := cmd.Run(); err != nil {
			return nil, err
		}
		jpegData = out.Bytes()
		
	default: // x11
		img, err := screenshot.CaptureDisplay(d.monitorIndex)
		if err != nil {
			return nil, err
		}
		
		var buf bytes.Buffer
		if err := jpeg.Encode(&buf, img, &jpeg.Options{Quality: d.quality}); err != nil {
			return nil, err
		}
		jpegData = buf.Bytes()
	}
	
	// Skip if unchanged (simple hash comparison)
	hash := simpleHash(jpegData)
	if hash == d.lastHash && len(jpegData) == len(d.lastFrame) {
		return nil, nil // unchanged
	}
	
	d.lastFrame = jpegData
	d.lastHash = hash
	
	return jpegData, nil
}

func simpleHash(data []byte) uint64 {
	var hash uint64 = 14695981039346656037
	for _, b := range data {
		hash ^= uint64(b)
		hash *= 1099511628211
	}
	return hash
}

func (i *InputController) ApplyEvent(event map[string]interface{}) {
	i.mu.Lock()
	defer i.mu.Unlock()
	
	eventType, _ := event["t"].(string)
	
	switch eventType {
	case "mmove":
		x := int(event["x"].(float64))
		y := int(event["y"].(float64))
		mouseMove(x, y)
		
	case "mdown":
		button := getMouseButton(event)
		mouseDown(button)
		
	case "mup":
		button := getMouseButton(event)
		mouseUp(button)
		
	case "mclick":
		x := int(event["x"].(float64))
		y := int(event["y"].(float64))
		button := getMouseButton(event)
		clicks := int(event["clk"].(float64))
		mouseMove(x, y)
		mouseClick(button, clicks)
		
	case "mwheel":
		dy := int(event["dy"].(float64))
		mouseScroll(dy)
		
	case "key":
		key := event["key"].(string)
		mods, _ := event["mod"].([]interface{})
		down, _ := event["down"].(bool)
		keyEvent(key, mods, down)
	}
}

	i.mu.Lock()
	defer i.mu.Unlock()
	
	for key := range i.heldKeys {
		keyUp(key)
	}
	i.heldKeys = make(map[string]bool)
}

	b, ok := event["b"].(float64)
	if !ok {
		return 1
	}
	return int(b)
}

	cmd := exec.Command("xdotool", "mousemove", "--sync",

	cmd := exec.Command("xdotool", "mouseup", string(rune('0'+button)))

	// xdotool uses button 4 for scroll up, 5 for scroll down
	button := 4
	if dy < 0 {
		button = 5
	}
	clicks := (dy + 99) / 100 // Convert to clicks
	if clicks < 0 {
		clicks = -clicks
	}
	for i := 0; i < clicks; i++ {
		cmd := exec.Command("xdotool", "click", string(rune('0'+button)))

		}
	}
}

func keyEvent(key string, mods []interface{}, down bool) {
	// Build xdotool key combination
	var keys []string
	
	// Add modifiers
	for _, mod := range mods {
		modStr, ok := mod.(string)
		if !ok {
			continue
		}
		switch strings.ToLower(modStr) {
		case "ctrl", "control":
			keys = append(keys, "ctrl")
		case "alt":
			keys = append(keys, "alt")
		case "shift":
			keys = append(keys, "shift")
		case "meta", "os", "cmd", "super":
			keys = append(keys, "super")
		}
	}
	
	// Map key name to xdotool key
	mappedKey := mapKey(key)
	if mappedKey != "" {
		keys = append(keys, mappedKey)
	}
	
	if len(keys) == 0 {
		return
	}
	
	keyCombo := strings.Join(keys, "+")
	
	if down {
		cmd := exec.Command("xdotool", "keydown", keyCombo)
		if err := cmd.Run(); err != nil {
			log.Printf("xdotool keydown failed: %v", err)
		}
	} else {
		cmd := exec.Command("xdotool", "keyup", keyCombo)
		if err := cmd.Run(); err != nil {
			log.Printf("xdotool keyup failed: %v", err)
		}
	}
}

func keyUp(key string) {
	mappedKey := mapKey(key)
	if mappedKey == "" {
		return
	}
	cmd := exec.Command("xdotool", "keyup", mappedKey)
	if err := cmd.Run(); err != nil {
		log.Printf("xdotool keyup failed: %v", err)
	}
}

func mapKey(key string) string {
	key = strings.ToLower(key)
	
	// Special keys mapping
	specialKeys := map[string]string{

	}

			return "F" + num
		}
	}
	
	// Single character
	if len(key) == 1 {
		return key
	}
	
	return ""
}

package main

import (
	"log"
	"syscall"
	"unsafe"

var (
	user32           = syscall.NewLazyDLL("user32.dll")
	procMouseMove    = user32.NewProc("SetCursorPos")
	procMouseEvent   = user32.NewProc("mouse_event")
	procSendInput    = user32.NewProc("SendInput")
	procVkKeyScan    = user32.NewProc("VkKeyScanW")
	procMapVirtualKey = user32.NewProc("MapVirtualKeyW")
)

const (
	MOUSEEVENTF_MOVE      = 0x0001
	MOUSEEVENTF_LEFTDOWN  = 0x0002
	MOUSEEVENTF_LEFTUP    = 0x0004
	MOUSEEVENTF_RIGHTDOWN = 0x0008
	MOUSEEVENTF_RIGHTUP   = 0x0010
	MOUSEEVENTF_MIDDLEDOWN = 0x0020
	MOUSEEVENTF_MIDDLEUP  = 0x0040
	MOUSEEVENTF_WHEEL     = 0x0800
	
	INPUT_MOUSE    = 0
	INPUT_KEYBOARD = 1
	
	KEYEVENTF_KEYDOWN = 0x0000
	KEYEVENTF_EXTENDEDKEY = 0x0001
	KEYEVENTF_KEYUP = 0x0002
)

type MOUSEINPUT struct {
	Dx          int32
	Dy          int32
	MouseData   uint32
	DwFlags     uint32
	Time        uint32
	DwExtraInfo uintptr
}

type KEYBDINPUT struct {
	WVk         uint16
	WScan       uint16
	DwFlags     uint32
	Time        uint32
	DwExtraInfo uintptr
}

type INPUT struct {
	Type uint32
	u    [40]byte // Union of MOUSEINPUT, KEYBDINPUT, HARDWAREINPUT
}

func mouseMove(x, y int) {
	ret, _, err := procMouseMove.Call(uintptr(x), uintptr(y))
	if ret == 0 {
		log.Printf("SetCursorPos failed: %v", err)
	}
}

func mouseDown(button int) {
	var flags uint32
	switch button {
	case 1:
		flags = MOUSEEVENTF_LEFTDOWN
	case 2:
		flags = MOUSEEVENTF_RIGHTDOWN
	case 3:
		flags = MOUSEEVENTF_MIDDLEDOWN
	}
	
	procMouseEvent.Call(uintptr(flags), 0, 0, 0, 0)
}

func mouseUp(button int) {
	var flags uint32
	switch button {
	case 1:
		flags = MOUSEEVENTF_LEFTUP
	case 2:
		flags = MOUSEEVENTF_RIGHTUP
	case 3:
		flags = MOUSEEVENTF_MIDDLEUP
	}
	
	procMouseEvent.Call(uintptr(flags), 0, 0, 0, 0)
}

func mouseClick(button, clicks int) {
	for i := 0; i < clicks; i++ {
		mouseDown(button)
		mouseUp(button)
	}
}

func mouseScroll(dy int) {
	clicks := dy / 100
	if clicks == 0 {
		clicks = 1
	}
	
	procMouseEvent.Call(uintptr(MOUSEEVENTF_WHEEL), 0, 0, uintptr(clicks*120), 0)
}

func keyEvent(key string, mods []interface{}, down bool) {
	// Press modifiers first
	for _, mod := range mods {
		modStr, ok := mod.(string)
		if !ok {
			continue
		}
		vk := getModifierVK(modStr)
		if vk != 0 {
			sendKeyEvent(vk, true)
		}
	}
	
	// Press main key
	vk := getVK(key)
	if vk != 0 {
		sendKeyEvent(vk, down)
	}
}

func keyUp(key string) {
	vk := getVK(key)
	if vk != 0 {
		sendKeyEvent(vk, false)
	}
}

func sendKeyEvent(vk uint16, down bool) {
	var input INPUT
	input.Type = INPUT_KEYBOARD
	
	kb := (*KEYBDINPUT)(unsafe.Pointer(&input.u[0]))
	kb.WVk = vk
	
	if down {
		kb.DwFlags = KEYEVENTF_KEYDOWN
	} else {
		kb.DwFlags = KEYEVENTF_KEYUP
	}
	
	procSendInput.Call(1, uintptr(unsafe.Pointer(&input)), unsafe.Sizeof(input))
}

func getVK(key string) uint16 {
	key = strings.ToLower(key)
	
	// Special keys
	specialKeys := map[string]uint16{
		"enter": 0x0D,
		"return": 0x0D,
		"backspace": 0x08,
		"tab": 0x09,
		"escape": 0x1B,
		"esc": 0x1B,
		"delete": 0x2E,
		"insert": 0x2D,
		"home": 0x24,
		"end": 0x23,
		"pageup": 0x21,
		"page_up": 0x21,
		"pagedown": 0x22,
		"page_down": 0x22,
		"capslock": 0x14,
		"space": 0x20,
		"arrowup": 0x26,
		"up": 0x26,
		"arrowdown": 0x28,
		"down": 0x28,
		"arrowleft": 0x25,
		"left": 0x25,
		"arrowright": 0x27,
		"right": 0x27,
		"printscreen": 0x2C,
		"print_screen": 0x2C,
	}
	
	if vk, ok := specialKeys[key]; ok {
		return vk
	}
	
	// Function keys
	if strings.HasPrefix(key, "f") && len(key) > 1 {
		num := key[1:]
		if num >= "1" && num <= "12" {
			n := int(num[0] - '0')
			if len(num) > 1 {
				n = n*10 + int(num[1]-'0')
			}
			return uint16(0x70 + n - 1)
		}
	}
	
	// Single character
	if len(key) == 1 {
		c := key[0]
		if c >= 'a' && c <= 'z' {
			return uint16('A' + c - 'a')
		}
		if c >= '0' && c <= '9' {
			return uint16(c)
		}
		// Use VkKeyScan for other characters
		ret, _, _ := procVkKeyScan.Call(uintptr(c))
		return uint16(ret & 0xFF)
	}
	
	return 0
}

func getModifierVK(mod string) uint16 {
	switch strings.ToLower(mod) {
	case "ctrl", "control":
		return 0x11 // VK_CONTROL
	case "alt":
		return 0x12 // VK_MENU
	case "shift":
		return 0x10 // VK_SHIFT
	case "meta", "os", "cmd", "super", "win":
		return 0x5B // VK_LWIN
	}
	return 0
}

func (d *DesktopBackend) captureScreen(width, height int) (image.Image, error) {
	hwnd, _, _ := procGetDesktopWindow.Call()
	hdc, _, _ := procGetDC.Call(hwnd)
	if hdc == 0 {
		return nil, fmt.Errorf("GetDC failed")
	}
	defer procReleaseDC.Call(hwnd, hdc)
	
	hdcMem, _, _ := procCreateCompatibleDC.Call(hdc)
	if hdcMem == 0 {
		return nil, fmt.Errorf("CreateCompatibleDC failed")
	}
	defer procDeleteDC.Call(hdcMem)
	
	hBitmap, _, _ := procCreateCompatibleBitmap.Call(hdc, uintptr(width), uintptr(height))
	if hBitmap == 0 {
		return nil, fmt.Errorf("CreateCompatibleBitmap failed")
	}
	defer procDeleteObject.Call(hBitmap)
	
	hOld, _, _ := procSelectObject.Call(hdcMem, hBitmap)
	defer procSelectObject.Call(hdcMem, hOld)
	
	ret, _, _ := procBitBlt.Call(hdcMem, 0, 0, uintptr(width), uintptr(height), hdc, 0, 0, uintptr(SRCCOPY))
	if ret == 0 {
		return nil, fmt.Errorf("BitBlt failed")
	}
	
	// Get bitmap data
	bmi := BITMAPINFO{
		BmiHeader: BITMAPINFOHEADER{
			BiSize:        uint32(unsafe.Sizeof(BITMAPINFOHEADER{})),
			BiWidth:       int32(width),
			BiHeight:      -int32(height), // Negative for top-down
			BiPlanes:      1,
			BiBitCount:    32,
			BiCompression: BI_RGB,
		},
	}
	
	dataSize := width * height * 4
	data := make([]byte, dataSize)
	
	ret, _, _ = procGetDIBits.Call(hdc, hBitmap, 0, uintptr(height),
		uintptr(unsafe.Pointer(&data[0])),
		uintptr(unsafe.Pointer(&bmi)),
		uintptr(DIB_RGB_COLORS))
	
	if ret == 0 {
		return nil, fmt.Errorf("GetDIBits failed")
	}
	
	// Convert BGRA to RGBA image
	img := image.NewRGBA(image.Rect(0, 0, width, height))
	for y := 0; y < height; y++ {
		for x := 0; x < width; x++ {
			offset := (y*width + x) * 4
			b := data[offset]
			g := data[offset+1]
			r := data[offset+2]
			a := data[offset+3]
			img.SetRGBA(x, y, color.RGBA{R: r, G: g, B: b, A: a})
		}
	}
	
	return img, nil
}

		Cols: uint16(s.cols),
	}

	ptmx, err := pty.StartWithSize(c.Cmd[0], c.Cmd[1:], &pty.Winsize{
		Rows: uint16(s.rows),
		Cols: uint16(s.cols),
	})
	if err != nil {
		return err
	}

	s.masterFd = ptmx
	s.cmd = ptmx.Process
	s.pgid = ptmx.Process.Pid

	return nil
}

func (s *ShellProcess) startPipes() error {
	// Windows implementation would go here
	return fmt.Errorf("Windows pipes not implemented yet")
}

func (s *ShellProcess) Resize(cols, rows int) {
	s.mu.Lock()
	defer s.mu.Unlock()
	
	s.cols = cols
	s.rows = rows
	
	if s.masterFd != nil && runtime.GOOS != "windows" {
		pty.Setsize(s.masterFd, &pty.Winsize{
			Rows: uint16(rows),
			Cols: uint16(cols),
		})
	}
}

func (s *ShellProcess) Read(buf []byte) (int, error) {
	if s.masterFd != nil {
		return s.masterFd.Read(buf)
	}
	return 0, fmt.Errorf("no master fd")
}

func (s *ShellProcess) Write(data []byte) (int, error) {
	if s.masterFd != nil {
		return s.masterFd.Write(data)
	}
	return 0, fmt.Errorf("no master fd")
}

func (s *ShellProcess) Alive() bool {
	if s.cmd == nil {
		return false
	}
	var status syscall.WaitStatus
	pid, err := syscall.Wait4(s.cmd.Pid, &status, syscall.WNOHANG, nil)
	if err != nil || pid == 0 {
		return true
	}
	return false
}

func (s *ShellProcess) Stop() {
	s.mu.Lock()
	defer s.mu.Unlock()

	if s.cmd == nil {
		return
	}

	if runtime.GOOS != "windows" && s.pgid > 0 {
		syscall.Kill(-s.pgid, syscall.SIGTERM)
		time.Sleep(3 * time.Second)
		syscall.Kill(-s.pgid, syscall.SIGKILL)
	} else {
		s.cmd.Kill()
	}

	if s.masterFd != nil {
		s.masterFd.Close()
		s.masterFd = nil
	}
}

// TerminalWorker handles terminal sessions
type TerminalWorker struct {
	ws      *websocket.Conn
	session SessionInfo
	onDone  func(string)
	shell   *ShellProcess
	stop    chan struct{}
}

func NewTerminalWorker(ws *websocket.Conn, session SessionInfo, onDone func(string)) *TerminalWorker {
	return &TerminalWorker{
		ws:      ws,
		session: session,
		onDone:  onDone,
		shell:   NewShellProcess(120, 30),
		stop:    make(chan struct{}),
	}
}

func (w *TerminalWorker) Run() {
	if err := w.shell.Start(); err != nil {
		log.Printf("Failed to start shell: %v", err)
		w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
			"t":      "bye",
			"reason": fmt.Sprintf("shell start failed: %v", err),
		}))
		w.ws.Close()
		w.onDone(w.session.ID)
		return
	}

	w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
		"t":    "hello",
		"mode": "terminal",
	}))

	go func() {
		buf := make([]byte, 65536)
		for {
			select {
			case <-w.stop:
				return
			default:
				n, err := w.shell.Read(buf)
				if err != nil || n == 0 {
					w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
						"t":      "bye",
						"reason": "shell exited",
					}))
					w.ws.Close()
					return
				}
				
				frame := append([]byte{TAG_TTY}, buf[:n]...)
				if err := w.ws.WriteMessage(websocket.BinaryMessage, frame); err != nil {
					return
				}
			}
		}
	}()

	go func() {
		for {
			select {
			case <-w.stop:
				return
			default:
				_, msg, err := w.ws.ReadMessage()
				if err != nil {
					return
				}
				
				if len(msg) == 0 {
					continue
				}
				
				tag := msg[0]
				payload := msg[1:]
				
				if tag == TAG_TTY {
					w.shell.Write(payload)
				} else if tag == TAG_CTRL {
					w.handleControl(payload)
				}
			}
		}
	}()
}

func (w *TerminalWorker) handleControl(payload []byte) {
	var msg map[string]interface{}
	if err := json.Unmarshal(payload, &msg); err != nil {
		return
	}

	if msg["t"] == "resize" {
		cols := int(msg["cols"].(float64))
		rows := int(msg["rows"].(float64))
		w.shell.Resize(cols, rows)
	}
}

func (w *TerminalWorker) Stop() {
	close(w.stop)
	w.shell.Stop()
	w.onDone(w.session.ID)
}

// DropOldestQueue is a thread-safe bounded queue that drops oldest items when full
type DropOldestQueue struct {
	items   [][]byte
	maxSize int
	mu      sync.Mutex
	cond    *sync.Cond
}

func NewDropOldestQueue(maxSize int) *DropOldestQueue {
	q := &DropOldestQueue{
		items:   make([][]byte, 0, maxSize),
		maxSize: maxSize,
	}
	q.cond = sync.NewCond(&q.mu)
	return q
}

func (q *DropOldestQueue) Put(item []byte) {
	q.mu.Lock()
	defer q.mu.Unlock()
	
	if len(q.items) >= q.maxSize {
		// Drop oldest
		q.items = q.items[1:]
	}
	q.items = append(q.items, item)
	q.cond.Signal()
}

func (q *DropOldestQueue) Get() []byte {
	q.mu.Lock()
	defer q.mu.Unlock()
	
	for len(q.items) == 0 {
		q.cond.Wait()
	}
	
	item := q.items[0]
	q.items = q.items[1:]
	return item
}

// DesktopWorker handles desktop sessions
type DesktopWorker struct {
	ws       *websocket.Conn
	session  SessionInfo
	onDone   func(string)
	backend  *DesktopBackend
	fps      int
	stop     chan struct{}
	frameQ   *DropOldestQueue
}

func NewDesktopWorker(ws *websocket.Conn, session SessionInfo, onDone func(string), monitor, fps, quality int) *DesktopWorker {
	return &DesktopWorker{
		ws:      ws,
		session: session,
		onDone:  onDone,
		backend: NewDesktopBackend(monitor, quality),
		fps:     fps,
		stop:    make(chan struct{}),
		frameQ:  NewDropOldestQueue(2),
	}
}

func (w *DesktopWorker) Run() {
	// Probe backend
	if err := w.backend.Probe(); err != nil {
		log.Printf("Desktop probe failed: %v", err)
		w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
			"t":      "bye",
			"reason": fmt.Sprintf("desktop probe failed: %v", err),
		}))
		w.ws.Close()
		w.onDone(w.session.ID)
		return
	}

	// Open backend
	if err := w.backend.Open(); err != nil {
		log.Printf("Desktop open failed: %v", err)
		w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
			"t":      "bye",
			"reason": fmt.Sprintf("desktop open failed: %v", err),
		}))
		w.ws.Close()
		w.onDone(w.session.ID)
		return
	}

	width, height := w.backend.Size()
	w.ws.WriteMessage(websocket.BinaryMessage, ctrl(map[string]interface{}{
		"t":      "hello",
		"mode":   "desktop",
		"width":  width,
		"height": height,
	}))

	// Capture goroutine
	go func() {
		interval := time.Second / time.Duration(w.fps)
		ticker := time.NewTicker(interval)
		defer ticker.Stop()

		for {
			select {
			case <-w.stop:
				return
			case <-ticker.C:
				jpeg, err := w.backend.CaptureJPEG()
				if err != nil {
					log.Printf("Capture error: %v", err)
					continue
				}
				if jpeg != nil {
					w.frameQ.Put(jpeg)
				}
			}
		}
	}()

	// Send frames goroutine
	go func() {
		for {
			select {
			case <-w.stop:
				return
			default:
				jpeg := w.frameQ.Get()
				if jpeg == nil {
					continue
				}
				frame := append([]byte{TAG_JPEG}, jpeg...)
				if err := w.ws.WriteMessage(websocket.BinaryMessage, frame); err != nil {
					return
				}
			}
		}
	}()

	// Receive control messages goroutine
	go func() {
		for {
			select {
			case <-w.stop:
				return
			default:
				_, msg, err := w.ws.ReadMessage()
				if err != nil {
					return
				}
				
				if len(msg) == 0 {
					continue
				}
				
				tag := msg[0]
				payload := msg[1:]
				
				if tag == TAG_CTRL {
					w.handleControl(payload)
				}
			}
		}
	}()
}

func (w *DesktopWorker) handleControl(payload []byte) {
	var msg map[string]interface{}
	if err := json.Unmarshal(payload, &msg); err != nil {
		return
	}

	eventType, _ := msg["t"].(string)
	
	switch eventType {
	case "quality":
		q := int(msg["q"].(float64))
		w.backend.SetQuality(q)
	default:
		w.backend.ApplyEvent(msg)
	}
}

func (w *DesktopWorker) Stop() {
	close(w.stop)
	w.backend.Close()
	w.onDone(w.session.ID)
}

// Agent is the main supervisor
type Agent struct {
	cfg         *Config
	control     *ControlClient
	workers     map[string]chan struct{}
	workersMu   sync.Mutex
	shutdown    chan struct{}
	lastAttach  map[string]time.Time
	lastAttachMu sync.Mutex
}

func NewAgent(cfg *Config) *Agent {
	return &Agent{
		cfg:        cfg,
		control:    NewControlClient(cfg.Server, cfg.AgentID),
		workers:    make(map[string]chan struct{}),
		shutdown:   make(chan struct{}),
		lastAttach: make(map[string]time.Time),
	}
}

func (a *Agent) Run() {
	log.Printf("pam_agent v%s | agent %s -> %s (terminal=%v desktop=%v)",
		AgentVersion, a.cfg.AgentID, a.cfg.Server, !a.cfg.NoTerminal, !a.cfg.NoDesktop)

	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		<-sigChan
		close(a.shutdown)
	}()

	go a.heartbeatLoop()
	go a.pollLoop()

	<-a.shutdown

	a.workersMu.Lock()
	for id, stop := range a.workers {
		close(stop)
		delete(a.workers, id)
	}
	a.workersMu.Unlock()

	a.control.post("/api/v1/agent/offline", map[string]string{"agent_id": a.cfg.AgentID})
	
	log.Println("agent stopped cleanly")
}

func (a *Agent) heartbeatLoop() {
	backoff := 1 * time.Second
	unknownWarned := false

	for {
		select {
		case <-a.shutdown:
			return
		default:
			res, err := a.control.Heartbeat()
			if err != nil {
				log.Printf("server unreachable (%v) — retrying in %v", err, backoff)
				time.Sleep(backoff)
				backoff *= 2
				if backoff > 30*time.Second {
					backoff = 30 * time.Second
				}
				continue
			}

			if ok, _ := res["ok"].(bool); !ok {
				if !unknownWarned {
					log.Printf("agent id %q is NOT registered — register this host in the PAM UI", a.cfg.AgentID)
					unknownWarned = true
				}
			} else {
				if unknownWarned {
					log.Printf("agent id %q now registered — online", a.cfg.AgentID)
					unknownWarned = false
				}
				backoff = 1 * time.Second
			}

			time.Sleep(HeartbeatInterval)
		}
	}
}

func (a *Agent) pollLoop() {
	for {
		select {
		case <-a.shutdown:
			return
		default:
			sessions, err := a.control.ActiveSessions()
			if err != nil {
				time.Sleep(PollInterval)
				continue
			}

			a.reconcile(sessions)
			time.Sleep(PollInterval)
		}
	}
}

func (a *Agent) reconcile(sessions []SessionInfo) {
	now := time.Now()

	for _, s := range sessions {
		a.workersMu.Lock()
		_, exists := a.workers[s.ID]
		a.workersMu.Unlock()

		if exists || s.Status != "active" {
			continue
		}

		a.lastAttachMu.Lock()
		lastTime := a.lastAttach[s.ID]
		a.lastAttachMu.Unlock()

		if now.Sub(lastTime) < AttachRetrySeconds {
			continue
		}

		if s.SessionType == "rdp" && a.cfg.NoDesktop {
			continue
		}
		if s.SessionType == "ssh" && a.cfg.NoTerminal {
			continue
		}

		a.workersMu.Lock()
		if len(a.workers) >= a.cfg.MaxSessions {
			a.workersMu.Unlock()
			log.Printf("max concurrent sessions reached — skipping %s", s.ID)
			continue
		}
		a.workersMu.Unlock()

		a.lastAttachMu.Lock()
		a.lastAttach[s.ID] = now
		a.lastAttachMu.Unlock()

		go a.runSession(s)
	}
}

func (a *Agent) runSession(session SessionInfo) {
	ws, err := a.control.SessionWS(session.ID)
	if err != nil {
		log.Printf("session %s connection error: %v", session.ID, err)
		return
	}
	defer ws.Close()

	stop := make(chan struct{})
	
	a.workersMu.Lock()
	a.workers[session.ID] = stop
	a.workersMu.Unlock()

	defer func() {
		a.workersMu.Lock()
		delete(a.workers, session.ID)
		a.workersMu.Unlock()
	}()

	var worker interface {
		Run()
		Stop()
	}

	if session.SessionType == "rdp" {
		worker = NewDesktopWorker(ws, session, func(id string) {}, a.cfg.Monitor, a.cfg.FPS, a.cfg.Quality)
	} else {
		worker = NewTerminalWorker(ws, session, func(id string) {})
	}

	go worker.Run()

	<-stop
	worker.Stop()

	log.Printf("%s session %s ended", session.SessionType, session.ID)
}
